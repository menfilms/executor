WPF-ГИДподключение этого UI к бэкенду на WPF .NET 8

01Проект и NuGet

Создаём обычный WPF-проект на .NET 8 и ставим пакет Microsoft.Web.WebView2. Файл этого интерфейса index.html просто кладём рядом с exe (в bin/…/net8.0-windows/).

dotnet new wpf -n Spectre
cd Spectre
dotnet add package Microsoft.Web.WebView2

# index.html копируем в папку проекта (окажется рядом с exe)

Если добавляешь пакет руками, в .csproj появится:

<ItemGroup>
  <PackageReference Include="Microsoft.Web.WebView2" Version="1.0.*" />
</ItemGroup>

02XAML: окно без рамки, WebView2 на всё окно

Убираем системную шапку (WindowStyle="None") и растягиваем WebView2 на всё окно — свой тайтлбар, кнопки и перетаскивание рисует сам веб-интерфейс. Размер окна зафиксирован (ResizeMode="NoResize"); если захочешь дать пользователю менять размер — верни CanResize.

<Window x:Class="Spectre.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        xmlns:wv2="clr-namespace:Microsoft.Web.WebView2.Wpf;assembly=Microsoft.Web.WebView2.Wpf"
        Title="SPECTRE" Width="1080" Height="680"
        WindowStyle="None" ResizeMode="NoResize"
        Background="#0b0e15" WindowStartupLocation="CenterScreen">
    <Grid>
        <!-- WebView2 растянут на всё окно -->
        <wv2:WebView2 x:Name="webView" DefaultBackgroundColor="#0b0e15"/>
    </Grid>
</Window>
DefaultBackgroundColor в цвет фона темы (--bg) убирает белую вспышку при загрузке.

03Инициализация WebView2 (MainWindow.xaml.cs)

Главное — зарегистрировать host-объект с именем spectre (имя уже зашито в index.html) до навигации. С этого момента JS видит его как chrome.webview.hostObjects.spectre и сам дёргает методы C#.

using System.IO;
using System.Threading.Tasks;
using System.Windows;
using Microsoft.Web.WebView2.Core;

namespace Spectre;

public partial class MainWindow : Window
{
    private CoreWebView2 _core;

    public MainWindow()
    {
        InitializeComponent();
        Loaded += async (_, _) => await Init();
    }

    private async Task Init()
    {
        await webView.EnsureCoreWebView2Async();
        _core = webView.CoreWebView2;
        // Защита от инспекции (раздел 07):
        _core.Settings.AreDefaultContextMenusEnabled = false; // без «Просмотреть код» по ПКМ
        _core.Settings.AreDevToolsEnabled = false;            // F12 вообще не откроется

        // 1. Мост: UI будет вызывать методы C# как spectre.Method(...)
        _core.AddHostObjectToScript("spectre", new SpectreBridge(this));

        // 2. Резервный канал UI -> C# (window.chrome.webview.postMessage)
        _core.WebMessageReceived += (_, e) =>
        {
            string raw = e.TryGetWebMessageAsString();
            // разбор на свой вкус, например через JsonDocument
        };

        // 3. Открываем интерфейс (index.html рядом с exe)
        var html = Path.Combine(AppContext.BaseDirectory, "index.html");
        await _core.NavigateToFileAsync(html);
    }

    // Отправка любой команды в UI
    public void Send(object obj) =>
        _core?.PostWebMessageAsString(
            System.Text.Json.JsonSerializer.Serialize(obj));
}

04Мост SpectreBridge — все кнопки сразу

Все кнопки интерфейса уже привязаны к методам этого класса — со стороны UI делать ничего не нужно, просто реализуй методы:

  • «Выполнить» (Ctrl+Enter) → Execute(lua) — приходит код активной вкладки
  • «Подключить / Отключить» → Attach(0) / Detach()
  • «Сохранить» → SaveFile(name, code)
  • Удаление файла → DeleteFile(name)
  • «Обновить файлы» → ListFiles() — в ответ отправь список (раздел 05)
  • «Свернуть» → Minimize(), «Закрыть» → CloseApp()
  • mousedown по тайтлбару → DragWindow() (раздел 06)
using System.IO;
using System.Runtime.InteropServices;
using System.Windows;

[ClassInterface(ClassInterfaceType.AutoDual)]
[ComVisible(true)]
public class SpectreBridge
{
    private readonly MainWindow _win;
    public SpectreBridge(MainWindow win) => _win = win;

    // ===== ВЫПОЛНИТЬ: сюда прилетает Lua активной вкладки =====
    public void Execute(string lua)
    {
        // сюда свой инжект (pipes / DLL / ...)
        _win.Send(new { t = "log", cls = "ok",
            msg = $"Получено {lua.Length} символов" });
    }

    // ===== ПОДКЛЮЧИТЬ / ОТКЛЮЧИТЬ =====
    public void Attach(int pid)
    {
        // Process.GetProcessesByName("RobloxPlayerBeta") ...
    }
    public void Detach() { }

    // ===== ФАЙЛЫ (папка Scripts рядом с exe) =====
    public void SaveFile(string name, string code)
        => File.WriteAllText(PathOf(name), code);

    public void DeleteFile(string name)
        => File.Delete(PathOf(name));

    public void ListFiles()
    {
        // собери список и отправь в UI (формат — раздел 05)
    }

    private string PathOf(string file)
    {
        var dir = Path.Combine(AppContext.BaseDirectory, "Scripts");
        Directory.CreateDirectory(dir);
        return Path.Combine(dir, Path.GetFileName(file)); // отрезаем ../
    }

    // ===== КНОПКИ ОКНА =====
    public void Minimize() => _win.WindowState = WindowState.Minimized;
    public void CloseApp() => _win.Close();

    // ===== ПЕРЕТАСКИВАНИЕ ОКНА =====
    public void DragWindow() => _win.DragMove();
}
Атрибуты ComVisible/AutoDual обязательны — без них WebView2 не увидит методы класса.

05Команды WPF → UI

Из C# в интерфейс шлём JSON через Send() (это PostWebMessageAsString). UI понимает 4 команды:

// Лог в консоль (cls: out | info | ok | warn | err | sys)
win.Send(new { t = "log", msg = "Инжект успешен", cls = "ok" });

// Открыть скрипт сразу во вкладке
win.Send(new { t = "open", name = "fly.lua", code = "print('hi')" });

// Синхронизировать список файлов (после SaveFile / ListFiles)
win.Send(new { t = "files", list = new[] {
    new { name = "fly.lua",  code = File.ReadAllText("Scripts/fly.lua") },
    new { name = "esp.lua",  code = "print('esp')" }
}});

// Сменить тему из C# (те же переменные, что в панели «Темы»)
win.Send(new { t = "theme", vars = new Dictionary<string, string> {
    ["--acc"] = "#7cff3f", ["--bg"] = "#070a07", ["--bg2"] = "#0c120c"
}});

Сериализатор сам экранирует кавычки и переводы строк в Lua-коде — собирать JSON руками не нужно.

Демо-режим: если интерфейс открыт в обычном браузере (без WebView2), он работает локально — вызовы моста игнорируются, файлы живут в localStorage. Ничего не ломается.

06Перетаскивание окна, когда WebView2 на весь Border

Системной шапки нет, всё окно — это WebView2, поэтому «тайтлбар» живёт внутри веб-страницы. Схема:

  • 1. У веб-интерфейса есть своя шапка (лого SPECTRE). Её mousedown вызывает spectre.DragWindow().
  • 2. В мосте DragWindow() делает _win.DragMove().
  • 3. DragMove() блокирует UI-поток до отпускания мыши — так и задумано: окно «прилипает» к курсору и следует за ним, пока кнопка не отпущена.
  • 4. Методы host-объекта уже выполняются в UI-потоке — Dispatcher не нужен.
  • 5. Размер окна зафиксирован (ResizeMode="NoResize") — интерфейс рассчитан на один размер и не поедет. Если нужно, верни CanResize, WebView2 растянется сам.
// уже есть в SpectreBridge:
public void DragWindow() => _win.DragMove();

// если вдруг вызываешь не из UI-потока:
public void DragWindow() =>
    _win.Dispatcher.Invoke(() => _win.DragMove());

Кнопки «Свернуть» и «Закрыть» в шапке работают по той же схеме — они уже вызывают Minimize() и CloseApp() из раздела 04.

07Анти-инспекция: правая кнопка и F12

Защита двухуровневая: на стороне C# инструменты отключаются на корню, на стороне JS (уже встроено в index.html) глушатся хоткеи и системное меню — работает даже в демо-режиме в браузере.

1. C# (основной уровень) — добавь при инициализации, уже есть в разделе 03:

_core.Settings.AreDefaultContextMenusEnabled = false; // ПКМ не покажет «Просмотреть код»
_core.Settings.AreDevToolsEnabled = false;            // F12 / хоткеи DevTools мертвы
_core.Settings.IsStatusBarEnabled = false;            // без строки состояния
_core.Settings.AreBrowserAcceleratorKeysEnabled = false; // Ctrl+W, Ctrl+T браузера не мешают

2. JS (уже в index.html) — перехватывает contextmenu, F12, Ctrl+Shift+I/J/C, Ctrl+U и крутит ловушку debugger: если инструменты всё же открыли, таймер замера зависает и факт фиксируется. Своё контекстное меню вкладок (ПКМ по вкладке) при этом продолжает работать — оно открывается своим обработчиком.

3. Дополнительно — можно полностью перебить window.open и скрыть следы окружения:

window.open = null;
// при желании — маскировать WebView2:
Object.defineProperty(chrome, 'webview', { get: () => undefined });
Важно: AreDevToolsEnabled = false — единственный надёжный способ. JS-перехват хоткеев легко обойти через меню, но меню у WebView2 нет, а системное ПКМ-меню уже отключено настройкой AreDefaultContextMenusEnabled.
